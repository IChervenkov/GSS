CREATE TABLE camps (
	id TEXT NOT NULL PRIMARY KEY,
	campname TEXT,
	created_at TIMESTAMP DEFAULT now()
);

CREATE TABLE users (
	id TEXT PRIMARY KEY NOT NULL,
	username TEXT,
	password TEXT
);

CREATE TABLE Soldier (
	id TEXT NOT NULL PRIMARY KEY,
	namesoldier TEXT,
	country TEXT,
	date_accommodation TIMESTAMP,
	date_free TIMESTAMP,
	meal_card TEXT,
	laundry_bag_id TEXT,
	used_room TEXT,
	camp_id TEXT,
	upcoming_accommodation DATE,
	upcoming_release DATE,
	FOREIGN KEY(laundry_bag_id) REFERENCES laundrybags(id),
	FOREIGN KEY(camp_id) REFERENCES camps(id)
);

	namebike TEXT,

	status_bike TEXT,

	FOREIGN KEY(soldierId) REFERENCES soldier(id),

CREATE TABLE rooms (

	FOREIGN KEY(buildId) REFERENCES buildings(id),

CREATE TABLE movesoldier (

	FOREIGN KEY(idpreviewkey) REFERENCES key(id),

CREATE TABLE assetstype (
	id INT NOT NULL PRIMARY KEY,

);

CREATE TABLE assets (
	id TEXT NOT NULL PRIMARY KEY,
	code TEXT,
	name_assets TEXT,
	type_id INT,
	location_room TEXT,
	location_key TEXT,
	categorie TEXT,
	quantity TEXT,
	mrah TEXT,
	asset_owner TEXT,
	status TEXT,
	expandable TEXT,
	description TEXT,
	camp_id TEXT,
	FOREIGN KEY(type_id) REFERENCES assetstype(id),
	FOREIGN KEY(location_room) REFERENCES rooms(id),
	FOREIGN KEY(location_key) REFERENCES key(id),
	FOREIGN KEY(camp_id) REFERENCES camps(id)
);

UPDATE soldier
SET used_room = res.id
FROM (
    SELECT k.id AS id, k.soldierid
    FROM key k
    WHERE k.soldierid IN (SELECT s.id FROM soldier s WHERE s.date_accommodation IS NOT NULL)
) AS res
WHERE soldier.id = res.soldierid;
